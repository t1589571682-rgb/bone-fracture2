    <script>
        const fileInput = document.getElementById('file-input');
        const showImg = document.getElementById('show-img');
        const resultArea = document.getElementById('result-area');

        // 图片上传监听
        fileInput.addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            // 图片预览
            const reader = new FileReader();
            reader.onload = function(res) {
                showImg.src = res.target.result;
                showImg.style.display = 'block';

                // 加载中提示 中韩双语
                resultArea.innerHTML = `
                    <p>영상 데이터를 분석 중입니다. 잠시만 기다려 주세요...</p>
                    <p class="result-kr">正在解析影像数据、提取特征，请稍候...</p>
                `;

                // 模拟分析延迟
                setTimeout(() => {
                    // 最终识别结果 双语对照
                    resultArea.innerHTML = `
                        <p>✅ 분석 완료 | 分析完成</p>
                        <p>▪ 영상 종류: 골격 X선 영상</p>
                        <p class="result-kr">▪ 影像类型：骨骼X光影像</p>

                        <p>▪ 판독 결과: 횡행 골절이 확인됨</p>
                        <p class="result-kr">▪ 判读结果：检出横行骨折</p>

                        <p>▪ 특징 설명: 안정성 골절로 골절선이 수평으로 형성되며, 단면 전위 위험이 낮습니다.</p>
                        <p class="result-kr">▪ 特征说明：属于稳定性骨折，骨折线呈水平走向，断端移位风险较低。</p>

                        <span class="status-tag">정상 판독 / 识别有效</span>
                    `;
                }, 1500);
            }
            reader.readAsDataURL(file);
        })
    </script>
